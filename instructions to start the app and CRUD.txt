View recent audit events from all three services:
SELECT occurred_at,
       service_name,
       actor_id,
       actor_role,
       operation,
       resource_type,
       resource_id,
       outcome,
       http_status,
       correlation_id
FROM (
    SELECT occurred_at, service_name, actor_id, actor_role, operation,
           resource_type, resource_id, outcome, http_status, correlation_id
    FROM public.auth_service_audit_events

    UNION ALL

    SELECT occurred_at, service_name, actor_id, actor_role, operation,
           resource_type, resource_id, outcome, http_status, correlation_id
    FROM public.job_service_audit_events

    UNION ALL

    SELECT occurred_at, service_name, actor_id, actor_role, operation,
           resource_type, resource_id, outcome, http_status, correlation_id
    FROM public.application_service_audit_events
) AS audit_events
ORDER BY occurred_at DESC
LIMIT 100;



To show only write events, add this condition to the outer query, before ORDER BY:
WHERE operation IN ('CREATE', 'UPDATE', 'DELETE')

Stop the microservices:

$expectedJars = @(
    'eureka-server-1.0.0.jar',
    'auth-service-1.0.0.jar',
    'job-service-1.0.0.jar',
    'application-service-1.0.0.jar',
    'gateway-1.0.0.jar'
)

$stoppedProcessIds = [System.Collections.Generic.HashSet[int]]::new()

Get-CimInstance Win32_Process -Filter "Name = 'java.exe'" |
    ForEach-Object {
        $processInfo = $_
        $jarMatch = [regex]::Match(
            [string]$processInfo.CommandLine,
            '(?i)-jar\s+"?([^"\s]+)'
        )

        if ($jarMatch.Success) {
            $jarName = [System.IO.Path]::GetFileName($jarMatch.Groups[1].Value)

            if ($expectedJars -contains $jarName) {
                $processIdValue = [int]$processInfo.ProcessId
                if ($stoppedProcessIds.Add($processIdValue)) {
                    Stop-Process -Id $processIdValue
                    Write-Host "Stopped $jarName (PID $processIdValue)."
                }
            }
        }
    }
